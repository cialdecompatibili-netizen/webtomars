(function () {
  var sec = document.getElementById('wm-space');
  if (!sec) return;
  var cv = sec.querySelector('canvas'), ctx = cv.getContext('2d');
  var box = sec.querySelector('.wm-orbitbox');
  var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var W = 0, H = 0, ox = 0, oy = 0, os = 300, lastW = 0;
  var stars = [], gals = [], planets = [];
  var m = { x: 0, y: 0, sx: -9999, sy: -9999, on: 0, target: 0 };
  var wp = { x: 0, y: 0, s: 0, d: 9999 };
  var TAU = Math.PI * 2, PULL = 240;
  function rnd(a, b) { return a + Math.random() * (b - a); }
  function clamp(v, a, b) { return v < a ? a : (v > b ? b : v); }
  function warp(x, y) {
    var dx = x - m.sx, dy = y - m.sy, d = Math.sqrt(dx * dx + dy * dy);
    wp.d = d;
    if (m.on < 0.01 || d >= PULL) { wp.x = x; wp.y = y; wp.s = 0; return; }
    var t = 1 - d / PULL, s = t * t * m.on;
    var nd = d * (1 - 0.9 * s), a = Math.atan2(dy, dx) + s * 1.7;
    wp.x = m.sx + nd * Math.cos(a); wp.y = m.sy + nd * Math.sin(a); wp.s = s; wp.d = nd;
  }
  function build() {
    var n = Math.min(Math.round(W * H / (W < 700 ? 4200 : 2600)), 520);
    stars = [];
    for (var i = 0; i < n; i++) stars.push({ x: Math.random(), y: Math.random(), z: rnd(0.25, 1), r: rnd(0.4, 1.5), ph: rnd(0, TAU), sp: rnd(0.8, 3), c: Math.random() < 0.18 ? 1 : (Math.random() < 0.15 ? 2 : 0) });
    var defs = [[0.1, 0.2, 0.2, '170,150,255', -0.5], [0.9, 0.78, 0.17, '255,140,110', 0.4], [0.64, 0.66, 0.12, '120,190,255', 0.9]];
    gals = [];
    defs.forEach(function (g) {
      var pts = [], N = W < 700 ? 150 : 260, rad = g[2] * Math.min(W, 900);
      for (var i = 0; i < N; i++) {
        var u = Math.pow(Math.random(), 0.65), arm = i % 2;
        var a = u * 3.6 + arm * Math.PI + rnd(-0.28, 0.28);
        pts.push({ r: u * rad, a: a, z: rnd(0.5, 1.4) });
      }
      gals.push({ cx: g[0], cy: g[1], rad: rad, tilt: g[4], col: g[3], pts: pts, sp: rnd(0.03, 0.06) });
    });
    var u = Math.min(W, 1100);
    planets = [
      { x: 0.93, y: 0.16, r: 0.022 * u + 8, k: 'ice', c1: '#d9f2ff', c2: '#4d7ab8', c3: '#0d1f4a' },
      { x: 0.58, y: 0.8, r: 0.03 * u + 8, k: 'ring', c1: '#ffe6b8', c2: '#c98d4a', c3: '#4a2a12' },
      { x: 0.55, y: 0.09, r: 0.013 * u + 6, k: 'moon', c1: '#f1f1f6', c2: '#8f8fa6', c3: '#2a2a3c' },
      { x: 0.84, y: 0.58, r: 0.02 * u + 8, k: 'band', c1: '#ffd3b0', c2: '#b8573a', c3: '#3a1410' }
    ];
  }
  function size() {
    var r = sec.getBoundingClientRect(), dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = r.width; H = r.height;
    cv.width = Math.round(W * dpr); cv.height = Math.round(H * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    var b = box.getBoundingClientRect();
    ox = b.left - r.left + b.width / 2; oy = b.top - r.top + b.height / 2; os = b.width;
    if (!stars.length || Math.abs(W - lastW) > 40) { build(); lastW = W; }
    if (reduce) draw(10);
  }
  function drawStars(T) {
    for (var i = 0; i < stars.length; i++) {
      var s = stars[i];
      var px = (s.x * W + T * 5 * s.z) % W, py = s.y * H;
      warp(px, py);
      if (wp.d < 14) continue;
      var tw = 0.55 + 0.45 * Math.sin(T * s.sp + s.ph);
      var al = clamp(tw * (0.35 + 0.65 * s.z) + wp.s * 0.9, 0, 1);
      if (wp.d < 40) al *= (wp.d - 14) / 26;
      var r = s.r * (0.6 + 0.6 * s.z) * (1 + wp.s * 1.8);
      ctx.fillStyle = s.c === 1 ? 'rgba(190,180,255,' + al + ')' : (s.c === 2 ? 'rgba(255,200,170,' + al + ')' : 'rgba(255,255,255,' + al + ')');
      if (r < 1.2) ctx.fillRect(wp.x, wp.y, r * 1.4, r * 1.4);
      else { ctx.beginPath(); ctx.arc(wp.x, wp.y, r, 0, TAU); ctx.fill(); }
    }
  }
  function drawGalaxies(T) {
    ctx.globalCompositeOperation = 'lighter';
    gals.forEach(function (g) {
      var cx = g.cx * W, cy = g.cy * H, cs = Math.cos(g.tilt), sn = Math.sin(g.tilt), rot = T * g.sp;
      warp(cx, cy);
      if (wp.d > 30) {
        var gr = ctx.createRadialGradient(wp.x, wp.y, 0, wp.x, wp.y, g.rad * 0.55);
        gr.addColorStop(0, 'rgba(' + g.col + ',0.45)'); gr.addColorStop(1, 'rgba(' + g.col + ',0)');
        ctx.fillStyle = gr; ctx.beginPath(); ctx.arc(wp.x, wp.y, g.rad * 0.55, 0, TAU); ctx.fill();
      }
      for (var i = 0; i < g.pts.length; i++) {
        var p = g.pts[i], a = p.a + rot;
        var lx = p.r * Math.cos(a), ly = p.r * Math.sin(a) * 0.42;
        warp(cx + lx * cs - ly * sn, cy + lx * sn + ly * cs);
        if (wp.d < 16) continue;
        var al = clamp(0.5 * (1 - p.r / g.rad * 0.5) + wp.s, 0.08, 1);
        ctx.fillStyle = 'rgba(' + g.col + ',' + al + ')';
        ctx.fillRect(wp.x, wp.y, p.z * (1 + wp.s * 2), p.z * (1 + wp.s * 2));
      }
    });
    ctx.globalCompositeOperation = 'source-over';
  }
  function sphere(x, y, r, c1, c2, c3) {
    var g = ctx.createRadialGradient(x - r * 0.4, y - r * 0.4, r * 0.1, x, y, r);
    g.addColorStop(0, c1); g.addColorStop(0.55, c2); g.addColorStop(1, c3);
    ctx.fillStyle = g; ctx.beginPath(); ctx.arc(x, y, r, 0, TAU); ctx.fill();
  }
  function drawPlanets(T) {
    planets.forEach(function (p, i) {
      var px = p.x * W + Math.sin(T * 0.05 + i) * 8, py = p.y * H + Math.cos(T * 0.04 + i) * 6;
      warp(px, py);
      if (wp.d < 26) return;
      var r = p.r * (1 - 0.35 * wp.s), a = wp.d < 60 ? (wp.d - 26) / 34 : 1;
      ctx.globalAlpha = 0.9 * a;
      if (p.k === 'ring') {
        ctx.save(); ctx.translate(wp.x, wp.y); ctx.rotate(-0.38);
        ctx.strokeStyle = 'rgba(255,225,170,.55)'; ctx.lineWidth = r * 0.22;
        ctx.beginPath(); ctx.ellipse(0, 0, r * 1.9, r * 0.5, 0, Math.PI, TAU); ctx.stroke();
        ctx.restore();
      }
      sphere(wp.x, wp.y, r, p.c1, p.c2, p.c3);
      if (p.k === 'band') {
        ctx.save(); ctx.beginPath(); ctx.arc(wp.x, wp.y, r, 0, TAU); ctx.clip();
        for (var b = -3; b <= 3; b++) { ctx.fillStyle = 'rgba(60,20,10,' + (b % 2 ? 0.16 : 0.08) + ')'; ctx.fillRect(wp.x - r, wp.y + b * r * 0.28 - r * 0.07, r * 2, r * 0.14); }
        ctx.restore();
      }
      if (p.k === 'ring') {
        ctx.save(); ctx.translate(wp.x, wp.y); ctx.rotate(-0.38);
        ctx.strokeStyle = 'rgba(255,225,170,.7)'; ctx.lineWidth = r * 0.22;
        ctx.beginPath(); ctx.ellipse(0, 0, r * 1.9, r * 0.5, 0, 0, Math.PI); ctx.stroke();
        ctx.restore();
      }
      ctx.globalAlpha = 1;
    });
  }
  function ringPoint(rad, a) { return [ox + rad * Math.cos(a), oy + rad * Math.sin(a)]; }
  function drawOrbits(T) {
    var rr = [200 / 480 * os, 135 / 480 * os, 72 / 480 * os];
    var delay = [0, 0.7, 1.4], alpha = [0.5, 0.75, 1], sat = [0.22, 0.38, 0.7], a0 = [0.8, 3.1, 5.2];
    for (var i = 0; i < 3; i++) {
      var p = clamp((T - delay[i]) / 1.5, 0, 1);
      if (p <= 0) continue;
      var N = 110, end = -Math.PI / 2 + TAU * p;
      ctx.strokeStyle = 'rgba(150,135,255,' + alpha[i] + ')'; ctx.lineWidth = 1.6; ctx.beginPath();
      for (var k = 0; k <= N; k++) {
        var a = -Math.PI / 2 + (end + Math.PI / 2) * k / N, q = ringPoint(rr[i], a);
        warp(q[0], q[1]);
        if (k === 0) ctx.moveTo(wp.x, wp.y); else ctx.lineTo(wp.x, wp.y);
      }
      ctx.stroke();
      if (p < 1) continue;
      var sa = a0[i] + T * sat[i];
      for (var t = 18; t >= 0; t--) {
        var qa = sa - t * 0.045, q2 = ringPoint(rr[i], qa);
        warp(q2[0], q2[1]);
        if (wp.d < 12) continue;
        var al = (1 - t / 19) * 0.9, sz = 1 + (1 - t / 19) * 3;
        ctx.fillStyle = 'rgba(190,180,255,' + al + ')';
        ctx.beginPath(); ctx.arc(wp.x, wp.y, sz, 0, TAU); ctx.fill();
      }
    }
  }
  function drawMars(T) {
    var p = clamp((T - 2.0) / 0.9, 0, 1);
    if (p <= 0) return;
